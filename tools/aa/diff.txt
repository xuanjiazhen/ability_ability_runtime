+long AbilityManagerShellCommand::ConvertTomorrowHourToTimestamp(int hour) {
+    // 1. 参数合法性校验
+    if (hour < 0 || hour > 23) {
+        std::cerr << "错误：小时参数必须是 0-23 之间的整数！" << std::endl;
+        return -1;
+    }
+
+    // 2. 获取当前系统时间（秒级）
+    auto now = std::chrono::system_clock::now();
+    time_t nowSec = std::chrono::system_clock::to_time_t(now);
+    tm localTime = *localtime(&nowSec);  // 转为本地时间（上海时区）
+
+    // 3. 调整为明天的指定小时（0分0秒）
+    localTime.tm_mday += 1;  // 日期 +1 → 明天
+    localTime.tm_hour = hour; // 设置指定小时
+    localTime.tm_min = 0;     // 分钟置0
+    localTime.tm_sec = 0;     // 秒置0
+
+    // 4. 转换为秒级时间戳，再转为毫秒级（long long 避免溢出）
+    time_t tomorrowSec = mktime(&localTime);
+    if (tomorrowSec == -1) {
+        std::cerr << "错误：日期转换失败（如月末日期+1超出当月天数）！" << std::endl;
+        return -1;
+    }
+    long long msTimestamp = (long long)tomorrowSec * 1000;
+
+    return msTimestamp;
+}
